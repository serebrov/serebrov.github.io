<div id="search"></div>
<script>
  window.addEventListener("DOMContentLoaded", () => {
    const ui = new PagefindUI({ element: "#search", showSubResults: true, showImages: false, autofocus: true });
    // The sidebar search form sends the query as ?q=...
    const query = new URLSearchParams(location.search).get("q");
    if (query) ui.triggerSearch(query);
  });
</script>
